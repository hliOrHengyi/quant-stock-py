{========== 知行多空线（参数14,28,57,114）+ N型上涨 + MACD零上 ==========}
{---------- 第一步：定义你的专属参数（完全照搬主图缺省值）----------}
M1:=14;
M2:=28;
M3:=57;
M4:=114;

{---------- 第二步：计算“知行多空线”（即你看到的黄线）----------}
{注意：必须用括号括起来，否则除法优先级错误}
多空线 := (MA(CLOSE, M1) + MA(CLOSE, M2) + MA(CLOSE, M3) + MA(CLOSE, M4)) / 4;

{---------- 第三步：计算 MACD 指标 ----------}
DIF := EMA(CLOSE, 12) - EMA(CLOSE, 26);
DEA := EMA(DIF, 9);
MACD := 2 * (DIF - DEA);

{---------- 条件1：MACD 在 0 轴之上----------}
COND_MACD := DIF > 0;

{---------- 条件2：K线 N型上涨（一波拉升→回调→再拉升）----------}
N1:=5;    {突破前高所用周期}
N2:=10;   {回调确认点}
N3:=20;   {第一波上涨起点}

上涨段 := REF(CLOSE, N2) > REF(CLOSE, N3);                     {20天前到10天前上涨}
回调段 := REF(CLOSE, N1) < REF(CLOSE, N2);                     {10天前到5天前回调}
再拉升 := CLOSE > REF(HHV(HIGH, N1), 1);                       {今日突破最近5日最高点}

COND_N := 上涨段 AND 回调段 AND 再拉升;

{---------- 条件3：20日内价格一直在“知行多空线”（黄线）之上 ----------}
COND_LINE := EVERY(CLOSE > 多空线, 20);                        {最近20个交易日收盘价均高于黄线}

{---------- 最终选股信号 ----------}
XG: COND_MACD AND COND_LINE;
