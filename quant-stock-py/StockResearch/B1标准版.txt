{===============================================================================
  第一部分：共享基础指标定义
  ===============================================================================}

{=== 知行趋势线体系 ===}
M1:=14;  {14日均线，约3周}
M2:=28;  {28日均线，约6周}
M3:=57;  {57日均线，约1个季度}
M4:=114; {114日均线，约2个季度}

知行短期趋势线:=EMA(EMA(C,10),10);  {白线：双重指数平滑，反映短期趋势方向}
知行多空线:=(MA(C,M1)+MA(C,M2)+MA(C,M3)+MA(C,M4))/4;  {黄线：四条均线均值，代表中长期价值中枢}

白线在黄线上:=知行短期趋势线 > 知行多空线;  {短期趋势强于长期价值中枢 → 多头}
当日收盘价达标:=C > 知行多空线;  {收盘价站在黄线之上}

{=== 砖型图（知行专有振荡器）===}
{原理：通过最高价与收盘价的关系构建多空力量对比}
VAR1A:=(HHV(H,4)-C)/(HHV(H,4)-LLV(L,4))*100-90;
VAR2A:=SMA(VAR1A,4,1)+100;
VAR3A:=(C-LLV(L,4))/(HHV(H,4)-LLV(L,4))*100;
VAR4A:=SMA(VAR3A,6,1);
VAR5A:=SMA(VAR4A,6,1)+100;
VAR6A:=VAR5A-VAR2A;
砖型图:=IF(VAR6A>4,VAR6A-4,0);  {红柱=多头占优，绿柱(缩小)=空头占优}

{砖型图状态判断}
前天绿柱:=REF(砖型图,2) < REF(砖型图,3);
昨天绿柱:=REF(砖型图,1) < REF(砖型图,2);
今天红柱:=砖型图 > REF(砖型图,1);
红柱高度:=砖型图 - REF(砖型图,1);
绿柱高度:=REF(砖型图,2) - REF(砖型图,1);
高度达标:=红柱高度 >= 绿柱高度*2/3;  {红柱力度至少达到绿柱的2/3，确保反转有力度}

{=== KDJ 随机指标 ===}
{参数：9日周期，标准KDJ计算}
RSV:=(C-LLV(L,9))/(HHV(H,9)-LLV(L,9))*100;
K:=SMA(RSV,3,1);
D:=SMA(K,3,1);
J:=3*K-2*D;  {J值可以超过100也可以低于0，负值即为"大负值"}

{=== MACD 趋势指标 ===}
{参数：12/26/9 标准MACD}
DIFF:=EMA(C,12)-EMA(C,26);
DEA:=EMA(DIFF,9);
MACD:=(DIFF-DEA)*2;  {红绿柱}
MACD在零轴上:=DIFF > 0;  {中长期趋势向上}

{=== 成交量分析 ===}
{20日、60日均量}
均量20:=MA(V,20);
均量60:=MA(V,60);
{地量判断：成交量处于近期最低水平}
地量20:=V <= LLV(V,20)*1.15;  {成交量不超过20日最低量的115%}
地量强:=V <= LLV(V,20)*1.08;  {更严格的地量标准}
{缩量趋势：成交量在萎缩中}
缩量趋势:=MA(V,3) < MA(V,5) AND MA(V,5) < MA(V,10);

{=== K线形态 ===}
{实体饱满：不是十字星或极小K线，上影线不能过长}
K线实体:=ABS(C-O)/(H-L+0.01);  {实体占比}
K线饱满:=K线实体 > 0.25;  {实体至少占25%}
上影线短:=(H-MAX(O,C))/(H-L+0.01) < 0.4;  {上影线不超过40%}

{=== 位置条件 ===}
{股价不要离黄线太远（低吸原则）}
接近黄线:=C/知行多空线 < 1.08;  {收盘价不超过黄线8%}
优选位置:=C/知行多空线 < 1.05;  {更严格：不超过5%}

{=== 不破前低 ===}
{前20个交易日最低价未被击穿，确保N型结构完好}
前低:=LLV(L,20);
不破前低:=L > 前低;
{更严格：N型结构的两个低点都在抬高}
N型低点抬高:=LLV(L,20) > LLV(L,40);  {20日最低点高于40日最低点，确认N型结构}


{===============================================================================
  B1 选股公式 — 标准版
  ================================================================================

  【适用场景】日常选股，捕捉标准的N型回调缩量低吸点。

  条件组合逻辑（对应Z哥B1五大条件）：
  ① N型上涨趋势（白线>黄线，收盘>黄线）
  ② 回调缩量到极致（成交量近20日地量）
  ③ KDJ J值勾到大负值反转（昨日J<5，今日J拐头向上）
  ④ 不破前一波N型结构低点
  ⑤ 在黄线附近（不超过黄线8%）

  + 砖型图绿转红确认（多空力量由空转多）
  + K线实体饱满（非十字星洗盘K线）
  + 上影线不能太长（无主力出货痕迹）
  ===============================================================================}

{--- B1_标准版 核心条件 ---}

{条件①：趋势向上 — 对应"处于N型上涨趋势中"}
B1_趋势:=白线在黄线上 AND 当日收盘价达标;

{条件②：极致缩量 — 对应"回调缩量到极致"}
B1_缩量:=地量20 AND 缩量趋势;

{条件③：KDJ大负值勾头 — 对应"J值勾到大负值"}
{昨日J值<5（已进入深坑/负数区域），今日J值开始向上勾头}
B1_KDJ:=REF(J,1) < 5 AND J > REF(J,1);

{条件④：砖型图反转 — 动量由空翻多的技术确认}
B1_砖型:=前天绿柱 AND 昨天绿柱 AND 今天红柱 AND 高度达标;

{条件⑤：位置合理 — 对应"在黄线/白线附近"}
B1_位置:=接近黄线;

{条件⑥：K线质量 — 实体饱满且无长上影（不是出货K线）}
B1_K线:=K线饱满 AND 上影线短;

{条件⑦：结构完整 — 不破前低}
B1_结构:=不破前低;

{--- B1 标准版 综合输出 ---}
B1_标准选股:= B1_趋势 AND B1_缩量 AND B1_KDJ AND B1_砖型 AND B1_位置 AND B1_K线 AND B1_结构;

XG: B1_标准选股;
